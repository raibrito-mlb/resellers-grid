Query 1 Funil Mensal por Canal
Visitas · Leads · Compras mensais por canal de origem → alimenta o array DATA[] (Aba Histórico Mensal)
meli-bi-data.MELIDATA.TRACKS SBOX_WHMPBR.RESELLERS_LEADS_TAN ~4-5 min · MELIDATA Atualizar: filtro ds BETWEEN 
WITH landing_raw AS (
  SELECT
    FORMAT_DATE('%Y-%m', t.ds)                                                    AS mes,
    t.usr.user_id                                                                   AS user_id,
    COALESCE(NULLIF(TRIM((SELECT v.value FROM UNNEST(t.platform.http.utm.map) AS v
                           WHERE v.key = 'utm_source' LIMIT 1)), ''), '')  AS src,
    COALESCE(NULLIF(TRIM((SELECT v.value FROM UNNEST(t.platform.http.utm.map) AS v
                           WHERE v.key = 'utm_medium' LIMIT 1)), ''), '')  AS med
  FROM `meli-bi-data.MELIDATA.TRACKS` t
  WHERE t.ds BETWEEN '2026-01-01' AND CURRENT_DATE()   -- ajustar intervalo ao atualizar
    AND t.bu = 'mercadopago'
    AND t.application.site_id = 'MLB'
    AND t.path = '/merchant_acquisition/flows/resellers/landing/home'
),
landing_origin AS (
  SELECT mes, user_id,
    CASE
      WHEN src = 'app' AND med = 'push'   THEN 'app / push'
      WHEN src = 'placement'              THEN 'placement / row_mgm'
      WHEN src = 'micrositio'             THEN 'micrositio'
      WHEN src = '' OR src = 'direct'     THEN 'direct'
      WHEN src = 'valueprop'              THEN 'valueprop / canva'
      WHEN LOWER(src) LIKE '%whatsapp%'   THEN 'whatsapp'
      ELSE 'outros'
    END AS canal
  FROM landing_raw
),
visitas AS (
  SELECT mes, canal, COUNT(DISTINCT user_id) AS visitas
  FROM landing_origin GROUP BY 1, 2
),
leads AS (
  SELECT lo.mes, lo.canal, COUNT(DISTINCT lo.user_id) AS leads
  FROM landing_origin lo
  INNER JOIN `meli-bi-data.SBOX_WHMPBR.RESELLERS_LEADS_TAN` l
    ON CAST(l.CUST AS STRING) = CAST(lo.user_id AS STRING)
    AND l.completo_cadastro IS NOT NULL
    AND FORMAT_DATE('%Y-%m', l.completo_cadastro) = lo.mes
  GROUP BY 1, 2
),
compras AS (
  SELECT lo.mes, lo.canal, COUNT(DISTINCT lo.user_id) AS compras
  FROM landing_origin lo
  INNER JOIN `meli-bi-data.SBOX_WHMPBR.RESELLERS_LEADS_TAN` l
    ON CAST(l.CUST AS STRING) = CAST(lo.user_id AS STRING)
    AND l.Completo_Venda IS NOT NULL
    AND FORMAT_DATE('%Y-%m', l.Completo_Venda) = lo.mes
  GROUP BY 1, 2
)
SELECT
  v.mes,
  v.canal,
  v.visitas,
  COALESCE(l.leads,   0) AS leads,
  COALESCE(c.compras, 0) AS compras,
  ROUND(SAFE_DIVIDE(COALESCE(l.leads,   0), v.visitas)                    * 100, 2) AS pct_leads_visitas,
  ROUND(SAFE_DIVIDE(COALESCE(c.compras, 0), COALESCE(l.leads, 0))       * 100, 2) AS pct_compras_leads,
  ROUND(SAFE_DIVIDE(COALESCE(c.compras, 0), v.visitas)                    * 100, 2) AS pct_compras_visitas
FROM visitas v
LEFT JOIN leads   l USING(mes, canal)
LEFT JOIN compras c USING(mes, canal)
ORDER BY v.mes, v.visitas DESC

Query 2 LP Diário por Canal
Visitas únicas à landing page por dia e por canal de origem → alimenta LP_DIARIO (Aba LP por Dia × Canal)
meli-bi-data.MELIDATA.TRACKS ~6-10 min · MELIDATA Atualizar: filtro ds BETWEEN 
WITH raw AS (
  SELECT
    CAST(t.ds AS STRING)                                                           AS dia,
    t.usr.user_id                                                                   AS user_id,
    COALESCE(NULLIF(TRIM((SELECT v.value FROM UNNEST(t.platform.http.utm.map) AS v
                           WHERE v.key = 'utm_source' LIMIT 1)), ''), '')  AS src,
    COALESCE(NULLIF(TRIM((SELECT v.value FROM UNNEST(t.platform.http.utm.map) AS v
                           WHERE v.key = 'utm_medium' LIMIT 1)), ''), '')  AS med
  FROM `meli-bi-data.MELIDATA.TRACKS` t
  WHERE t.ds BETWEEN '2026-01-01' AND CURRENT_DATE()   -- ajustar intervalo ao atualizar
    AND t.bu = 'mercadopago'
    AND t.application.site_id = 'MLB'
    AND t.path = '/merchant_acquisition/flows/resellers/landing/home'
)
SELECT
  dia,
  CASE
    WHEN src = 'app' AND med = 'push'   THEN 'app / push'
    WHEN src = 'placement'              THEN 'placement / row_mgm'
    WHEN src = 'micrositio'             THEN 'micrositio'
    WHEN src = '' OR src = 'direct'     THEN 'direct'
    WHEN src = 'valueprop'              THEN 'valueprop / canva'
    WHEN LOWER(src) LIKE '%whatsapp%'   THEN 'whatsapp'
    ELSE 'outros'
  END AS canal,
  COUNT(DISTINCT user_id) AS visitas
FROM raw
GROUP BY 1, 2
ORDER BY 1, 3 DESC

Query 3 Funil Direto LEADS_TAN
Cadastros × 1ª compra por mês (3 sub-queries) → alimenta FD[] (Aba Funil Direto)
SBOX_WHMPBR.RESELLERS_LEADS_TAN ~5-10 seg · sandbox 3 queries independentes
3a — Cadastrados por mês
SELECT
  FORMAT_DATE('%Y-%m', completo_cadastro) AS mes,
  COUNT(DISTINCT CUST)                    AS cadastrados
FROM `meli-bi-data.SBOX_WHMPBR.RESELLERS_LEADS_TAN`
WHERE completo_cadastro BETWEEN '2026-01-01' AND CURRENT_DATE()
GROUP BY 1
ORDER BY 1
3b — 1ª Compra no mesmo mês do cadastro
SELECT
  FORMAT_DATE('%Y-%m', completo_cadastro) AS mes,
  COUNT(DISTINCT CUST)                    AS compra_mesmo_mes
FROM `meli-bi-data.SBOX_WHMPBR.RESELLERS_LEADS_TAN`
WHERE completo_cadastro BETWEEN '2026-01-01' AND CURRENT_DATE()
  AND Completo_Venda IS NOT NULL
  AND FORMAT_DATE('%Y-%m', Completo_Venda) = FORMAT_DATE('%Y-%m', completo_cadastro)
GROUP BY 1
ORDER BY 1
3c — 1ª Compra realizada no mês (qualquer safra)
SELECT
  FORMAT_DATE('%Y-%m', Completo_Venda) AS mes,
  COUNT(DISTINCT CUST)                 AS compra_indep
FROM `meli-bi-data.SBOX_WHMPBR.RESELLERS_LEADS_TAN`
WHERE Completo_Venda BETWEEN '2026-01-01' AND CURRENT_DATE()
GROUP BY 1
ORDER BY 1
