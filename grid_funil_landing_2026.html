WITH lp_raw AS (
  SELECT t.ds, t.usr.user_id AS user_id,
    COALESCE(NULLIF(TRIM((SELECT v.value FROM UNNEST(t.platform.http.utm.map) AS v WHERE v.key='utm_source' LIMIT 1)),''),'') AS src,
    COALESCE(NULLIF(TRIM((SELECT v.value FROM UNNEST(t.platform.http.utm.map) AS v WHERE v.key='utm_medium' LIMIT 1)),''),'') AS med,
    t.device.os_name AS os, t.device.platform AS device
  FROM `meli-bi-data.MELIDATA.TRACKS` t
  WHERE t.ds BETWEEN '{start}' AND '{end}'
    AND t.bu='mercadopago' AND t.application.site_id='MLB'
    AND t.path='/merchant_acquisition/flows/resellers/landing/home'
)
SELECT ds,
  CASE WHEN src='app' AND med='push' THEN 'app / push'
       WHEN src='placement' THEN 'placement / row_mgm'
       WHEN src='micrositio' THEN 'micrositio'
       WHEN src='home_mp' THEN 'banner_xsell'
       WHEN src='' OR src='direct' THEN 'direct'
       WHEN src='valueprop' THEN 'valueprop / canva'
       WHEN LOWER(src) LIKE '%whatsapp%' THEN 'whatsapp'
       ELSE 'outros' END AS canal,
  COALESCE(os,'desconhecido') AS os,
  COALESCE(device,'desconhecido') AS device,
  COUNT(DISTINCT user_id) AS visitas
FROM lp_raw GROUP BY 1,2,3,4 ORDER BY 1,2

      WHEN '/merchant_acquisition/flows/resellers/store/selection' THEN '4_loc_selecao'

Q_PERFIL Perfil por experiência e modelo de operação, com cadastros, ativação M0 e TPV M0 nos últimos seis meses fechados. monthly SBOX_WHMPBR.RESELLERS_LEADS_TAN ▼
Statusem uso
TabelasSBOX_WHMPBR.RESELLERS_LEADS_TANWHOWNER.LK_MP_SGROWTH_RESELLERS_TRIAL_FORM_RESPONSESWHOWNER.BT_PRODUCT_ORDERSWHOWNER.BT_PRODUCT_ORDER_ITEMWHOWNER.LK_PRODUCT_DEVICEWHOWNER.BT_MP_IC_CUBO_SELLERS
WITH
base AS (
  SELECT
    DATE_TRUNC(DATE(l.completo_cadastro), MONTH) AS mes,
    CAST(l.CUST AS STRING) AS user_id,
    DATE(l.completo_cadastro) AS cadastro_date,
    (l.Completo_Venda IS NOT NULL) AS ativado,
    (l.Completo_Venda IS NOT NULL
       AND DATE_DIFF(DATE(l.Completo_Venda), DATE(l.completo_cadastro), DAY) BETWEEN 0 AND 30) AS ativo_m0
  FROM `meli-bi-data.SBOX_WHMPBR.RESELLERS_LEADS_TAN` l
  WHERE l.completo_cadastro IS NOT NULL
    AND DATE(l.completo_cadastro) BETWEEN '{start}' AND '{end}'
),
perfil AS (
  SELECT CAST(user_id AS STRING) AS user_id,
    CASE WHEN STARTS_WITH(LOWER(experiencia_raw),'especialista') THEN 'Especialista'
         WHEN STARTS_WITH(LOWER(experiencia_raw),'intermedi')    THEN 'Intermediário'
         WHEN STARTS_WITH(LOWER(experiencia_raw),'iniciante')    THEN 'Iniciante' ELSE 'Outros' END AS experiencia,
    CASE WHEN STARTS_WITH(LOWER(modelo_op_raw),'gestor')    THEN 'Gestor de Equipe'
         WHEN STARTS_WITH(LOWER(modelo_op_raw),'estrat')    THEN 'Estratégia Digital'
         WHEN STARTS_WITH(LOWER(modelo_op_raw),'indica')    THEN 'Indicação de Balcão'
         WHEN STARTS_WITH(LOWER(modelo_op_raw),'consultor') THEN 'Consultor Autônomo' ELSE 'Outros' END AS modelo_op
  FROM (
    SELECT f.user_id,
      MAX(CASE WHEN f.QUESTION_ORDER=1 THEN f.ANSWER_TEXT END) AS experiencia_raw,
      MAX(CASE WHEN f.QUESTION_ORDER=2 THEN f.ANSWER_TEXT END) AS modelo_op_raw
    FROM `meli-bi-data.WHOWNER.LK_MP_SGROWTH_RESELLERS_TRIAL_FORM_RESPONSES` f
    GROUP BY f.user_id
  )
),
ativos_m0 AS (SELECT user_id FROM base WHERE ativo_m0),
device_orders AS (
  SELECT CAST(o.CUS_CUST_ID_BUY AS STRING) AS user_id, d.PAY_MPOS_DEVICE_ID
  FROM `meli-bi-data.WHOWNER.BT_PRODUCT_ORDERS` o
  JOIN `meli-bi-data.WHOWNER.BT_PRODUCT_ORDER_ITEM` oi ON o.IDX_ID=oi.IDX_ID
  JOIN `meli-bi-data.WHOWNER.LK_PRODUCT_DEVICE` d ON oi.PD_ID=d.PD_ID
  JOIN ativos_m0 a ON CAST(o.CUS_CUST_ID_BUY AS STRING)=a.user_id
  WHERE o.ORDER_GATEWAY='1' AND o.SIT_SITE_ID='MLB' AND o.OPERATION_TYPE='1' AND oi.PI_PRODUCT_TYPE=1
),
tpv_30d AS (
  SELECT b.user_id, SUM(c.TPV_EM_REAIS) AS tpv_m0
  FROM base b
  JOIN device_orders do ON b.user_id=do.user_id
  JOIN `meli-bi-data.WHOWNER.BT_MP_IC_CUBO_SELLERS` c
    ON c.PAY_MPOS_DEVICE_ID=do.PAY_MPOS_DEVICE_ID
   AND c.TIM_DAY BETWEEN b.cadastro_date AND DATE_ADD(b.cadastro_date, INTERVAL 30 DAY)
   AND c.TIM_DAY BETWEEN '{start}' AND DATE_ADD(DATE '{end}', INTERVAL 30 DAY)
  WHERE b.ativo_m0
  GROUP BY 1
),
joined AS (
  SELECT b.mes, p.experiencia, p.modelo_op, b.user_id, b.ativado, b.ativo_m0, COALESCE(t.tpv_m0,0) AS tpv_m0
  FROM base b
  JOIN perfil p ON b.user_id=p.user_id
  LEFT JOIN tpv_30d t ON b.user_id=t.user_id
  WHERE p.experiencia<>'Outros' AND p.modelo_op<>'Outros'

SELECT

  COUNT(DISTINCT IF(ativado, user_id, NULL)) AS ativados,

WITH tracks AS (
  SELECT ds, usr.user_id AS user_id,
    JSON_VALUE(event_data, '$.entrypoint') AS entrypoint,
    JSON_VALUE(event_data, '$.custom')      AS custom
  FROM `meli-bi-data.MELIDATA.TRACKS`
  WHERE ds BETWEEN '{start}' AND '{end}'
    AND path LIKE '/merchant_acquisition%'
),
classified AS (
  SELECT ds, user_id,
    CASE
      WHEN entrypoint IS NULL                                                        THEN '1_visita_lp'
      WHEN entrypoint='questionnaire-option-selected' AND custom LIKE '%EXPERIENCE_MLB%'    THEN '2_profiling_exp'
      WHEN entrypoint='questionnaire-option-selected' AND custom LIKE '%BUSINESS_MLB%'      THEN '3_tipo_negocio'
      WHEN entrypoint='questionnaire-option-selected' AND custom LIKE '%USER%'              THEN '4_loc_selecao'
      WHEN entrypoint='questionnaire-confirm-button'  AND custom LIKE '%USER_LOCATION_MLB%' THEN '5_loc_confirmacao'
      ELSE NULL
    END AS step
  FROM tracks
),
funnel_tracks AS (SELECT ds, user_id, step FROM classified WHERE step IS NOT NULL),

  SELECT DATE(completo_cadastro) AS ds, CAST(CUST AS STRING) AS user_id, '6_cadastro_concluido' AS step
